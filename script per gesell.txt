% nulla di anomalo dal confronto tra residui e variabile esplicativa - fine della diagnostica sulla componente parametrica;

diagnostica sulla compatibilità dei dati;

proc reg data = gesell;
model y = x/ r influence;
output out = uno r= res1 p = pred1 h = h1 cookd = cook1;
run;

proc sgscatter data = uno;
plot h1*cook1/datalabel = cod;
run;

% tolgo la 18 che ha leverage e Cook pèiù elevati di tutti;
